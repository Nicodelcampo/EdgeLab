// ============================================================================
// HFTZones2.cs — v2.0 — Detector de ráfagas HFT multi-instrumento ADAPTATIVO
// Consolida: HFTZonesESPureV2 (motor + FIX#1-3) + HFTZonesNQPureV3 (retro relativo)
// Piloto del puente NT8 → EdgeLab/vectorbt. Sigue la Guía reutilizable §1–14.
//
// ============================== CONTRATO ====================================
// DEFINICIÓN DEL EVENTO
//   Racha de trades consecutivos en una dirección sobre la subserie 1-tick.
//   Un tick es válido si continúa la dirección o es plano (mismo precio).
//   La racha NO se inicia en ticks planos (v1 iniciaba con isDown-first =
//   sesgo bajista arbitrario — defecto corregido y declarado).
//   Corta por: (a) fallos > FallosTolerados; (b) pausa > MaxPausaMs efectivo;
//   (c) retroceso relativo > max(RetroFloorTicks, RetroPctHeight% × altura)
//   (si UseRelativeRetro; el tick que viola queda INCLUIDO en la racha);
//   (d) límite de sesión. Al cortar se evalúa y clasifica (Finalizar).
//
// FIXES DE V1 PRESERVADOS (declarados)
//   FIX#1 duración con piso 1 ms para vol_rate (ráfagas sub-ms).
//   FIX#2 avg_ms solo sobre intervalos INTERNOS de la racha (excluye el gap
//         previo al primer tick).
//   FIX#3 sweep vs absorción: sweep si altura >= MinSweepTicks; si no y
//         DetectAbsorb, candidata ABSORB con MinAbsorbPasos.
//
// CALIBRACIÓN ADAPTATIVA (lo nuevo — multi-instrumento)
//   Principio (guía §14): los parámetros ESTRUCTURALES (MinPasos, MinSweepTicks,
//   FallosTolerados, retro) son la hipótesis y quedan FIJOS y declarados.
//   Solo se adaptan las ESCALAS de instrumento: velocidad (ms) y volumen.
//   - Ventana de calibración = la SESIÓN ANTERIOR completa. La calibración se
//     CONGELA al inicio de cada sesión (causal, reproducible; nunca rolling
//     intrabar — evita que la misma racha pase/falle según el momento del día).
//   - Muestras: intervalos inter-tick en ms con 0 <= ms <= PauseExcludeMs
//     (excluye pausas) y volumen por tick. Decimación DETERMINISTA por stride
//     con duplicación al llegar a CalibSampleCap (nunca muestreo aleatorio).
//   - Cuantiles EXACTOS sin interpolación: idx = ceil(q*n)-1 (igual que
//     VolTicksPOC2) — replicable 1:1 en numpy.
//   - FÓRMULAS (declaradas; todas exportadas en el evento CALIBRATION):
//       eff_predator_ms = max(1, Q(QPredator))            [def q=0.02]
//       eff_ultra_ms    = max(eff_predator_ms, Q(QUltra)) [def q=0.05]
//       eff_max_avg_ms  = max(eff_ultra_ms, Q(QMaxAvg))   [def q=0.15]
//       eff_max_pausa_ms= clamp(PauseMult × max(1,Q(0.50)), eff_max_avg_ms, 5000)
//       eff_max_total_ms= eff_max_avg_ms × MinPasos × TotalMsMult
//       eff_min_total_vol = VolMultMedianTick × median(vol_tick) × MinPasos
//       eff_min_vol_rate  = eff_min_total_vol / (eff_max_total_ms/1000)
//   - AUDITORÍA DE RESOLUCIÓN (gate P0): CALIBRATION exporta frac_zero_ms y
//     n; si Q(0.50) <= 0 ⇒ resolution_limited=1 (los umbrales de velocidad
//     quedan con piso 1 ms y NO son confiables — verificar timestamps ms
//     reales del feed antes de usar buckets de velocidad).
//   - Primera sesión del dataset: sin calibración ⇒ SIN detección (evento
//     CALIBRATION_PENDING). Sin fallbacks silenciosos.
//   - AdaptiveMode=false ⇒ usa los parámetros manuales del grupo C (para A/B
//     contra las v1 por instrumento). calib_id=0.
//   - Cada zona exporta el calib_id usado ⇒ la fuerza bruta offline puede
//     reconstruir exactamente qué umbrales generaron cada zona.
//
// CLASIFICACIÓN (post-filtros, jerárquica, declarada)
//   ABSORB si altura < MinSweepTicks; si no por velocidad:
//   PREDATOR avg_ms <= eff_predator_ms; ULTRA <= eff_ultra_ms; FAST resto.
//
// GEOMETRÍA DE ZONA (corrige inconsistencia v1: la zona bear se dibujaba
//   [swH-tick, swH] pero se mitigaba arriba de swH+tick)
//   Racha alcista (barre hacia arriba): origen = swL → zona SOPORTE
//     [swL − ZoneHeightTicks×TickSize, swL].
//   Racha bajista: origen = swH → zona RESISTENCIA [swH, swH + ZoneHeightTicks×TickSize].
//
// CICLO DE VIDA (tick-exacto)
//   ZONE_CREATED → ZONE_TOUCHED por época (precio entra a [lower,upper]
//   estando afuera) → ZONE_INVALIDATED reason ∈ {first_touch, close_through,
//   max_touches} según InvalidationMode/MaxTouches → ZONE_EXPIRED por
//   MaxAgeBars (barras primarias; chart canónico). close_through: precio
//   supera el borde distal en PenetrationTicks. Las zonas NO viven para
//   siempre (v1: lista sin límite, sin expiración = defecto).
//
// EXPORT (stream de EVENTOS — nunca dump de estado final; v1 volcaba el CSV
//   completo en Terminated con start_ms==end_ms — defecto)
//   '# meta' + '# params' + filas: OBS (toda racha finalizada con
//   valid_steps >= MinExportValidSteps, con flag pass=0/1 → permite barrer
//   umbrales offline sin re-correr NT8), ZONE_CREATED/TOUCHED/INVALIDATED/
//   EXPIRED, CALIBRATION, CALIBRATION_PENDING, SESSION_END, ERROR.
//   Sin rutas hardcodeadas (v1: C:\AlgoNQ\..., C:\Users\nicoc\... — defecto);
//   sin SQLite en el indicador: el Store ingiere el CSV.
//
// THREADING
//   Detección en el hilo de la subserie tick; Draw.* SOLO al cierre de barra
//   primaria (v1 dibujaba desde el hilo tick — defecto).
//
// LO QUE NO HACE (offline en EdgeLab): contexto de sesión/news, decaimiento
//   por toques como señal (acá solo visual), joins con otras zonas.
//
// PARIDAD (gates): P0 ticks + resolución ms → P2A creación → P2B lifecycle.
// ============================================================================

		protected override void OnBarUpdate()
		{
			// ── Subserie 1-tick ──
			if (BarsInProgress == 1)
			{
				if (CurrentBars[1] < 1) return;
​
				DateTime t = Times[1][0];
				CheckSession(t);
​
				double price = Closes[1][0];
				double vol   = Volumes[1][0];
				double ms    = (Times[1][0] - Times[1][1]).TotalMilliseconds;
​
				// Muestreo para calibrar la PRÓXIMA sesión (siempre activo)
				if (AdaptiveMode)
				{
					if (ms >= 0 && ms <= PauseExcludeMs)
						AddSample(_msSamples, ms, ref _msStride, ref _msSeen);
					AddSample(_volSamples, vol, ref _volStride, ref _volSeen);
				}
​
				// Ciclo de vida tick-exacto ANTES de detectar
				UpdateZonesWithTick(price, t);
​
				if (_calibReady)
					StepEngine(price, vol, ms, t);
				else if (!_pendingLogged)
				{
					LogRow("CALIBRATION_PENDING", t, null, "first_session_no_calibration", "");
					_pendingLogged = true;
				}
				return;
			}
​
			// ── Serie primaria: expiración + render ──
			if (BarsInProgress != 0) return;
			if (CurrentBars[0] < 1) return;
			ExpireZones();
			DrawZones();
		}

		#region Sesiones y calibración
		private void CheckSession(DateTime t)
		{
			if (_sessEnd != DateTime.MinValue && t < _sessEnd) return;
​
			_sessIter.GetNextSession(t, true);
			_sessEnd = _sessIter.ActualSessionEnd;
​
			// El límite de sesión corta cualquier racha en curso
			FinalizeStreak(t);
​
			if (!AdaptiveMode) return;
​
			if (_msSamples.Count >= 100 && _volSamples.Count >= 100 && _msSeen >= MinCalibSamples)
				Recalibrate(t);
			// Si no alcanza: se mantiene la calibración anterior (o sigue pending)
​
			_msSamples.Clear();  _volSamples.Clear();
			_msStride = 1; _volStride = 1; _msSeen = 0; _volSeen = 0;
		}
​
		private void Recalibrate(DateTime t)
		{
			List<double> ms  = new List<double>(_msSamples);  ms.Sort();
			List<double> vv  = new List<double>(_volSamples); vv.Sort();
​
			double qPred = Quantile(ms, QPredator);
			double qUlt  = Quantile(ms, QUltra);
			double qMax  = Quantile(ms, QMaxAvg);
			double p50   = Quantile(ms, 0.50);
			double medV  = Quantile(vv, 0.50);
​
			int zeroCount = 0;
			for (int i = 0; i < ms.Count && ms[i] <= 0; i++) zeroCount++;
			double fracZero = ms.Count > 0 ? (double)zeroCount / ms.Count : 1.0;
​
			_effPredMs      = Math.Max(1.0, qPred);
			_effUltraMs     = Math.Max(_effPredMs, qUlt);
			_effMaxAvgMs    = Math.Max(_effUltraMs, qMax);
			_effMaxPausaMs  = Math.Min(5000.0, Math.Max(_effMaxAvgMs, PauseMult * Math.Max(1.0, p50)));
			_effMaxTotalMs  = _effMaxAvgMs * MinPasos * TotalMsMult;
			_effMinTotalVol = VolMultMedianTick * medV * MinPasos;
			_effMinVolRate  = _effMinTotalVol / (_effMaxTotalMs / 1000.0);
			_resLimited     = p50 <= 0.0;
			_calibId++;
			_calibReady     = true;
​
			string extra = string.Format(CultureInfo.InvariantCulture,
				"n_ms={0};n_vol={1};stride_ms={2};frac_zero_ms={3:F4};p50_ms={4:F2};median_tick_vol={5:F2};eff_predator_ms={6:F2};eff_ultra_ms={7:F2};eff_max_avg_ms={8:F2};eff_max_pausa_ms={9:F2};eff_max_total_ms={10:F2};eff_min_total_vol={11:F2};eff_min_vol_rate={12:F2};resolution_limited={13}",
				ms.Count, vv.Count, _msStride, fracZero, p50, medV,
				_effPredMs, _effUltraMs, _effMaxAvgMs, _effMaxPausaMs, _effMaxTotalMs,
				_effMinTotalVol, _effMinVolRate, _resLimited ? 1 : 0);
			LogRow("CALIBRATION", t, null, "", extra);
			if (_resLimited)
				Print("[HFTZones2] ATENCIÓN: resolución de timestamps limitada (p50<=0ms). Buckets de velocidad NO confiables. Verificar gate P0.");
		}

		// Cuantil exacto sin interpolación (replicable 1:1 offline)
		private static double Quantile(List<double> sorted, double q)
		{
			int n = sorted.Count;
			if (n == 0) return 0;
			int idx = (int)Math.Ceiling(q * n) - 1;
			if (idx < 0) idx = 0;
			if (idx > n - 1) idx = n - 1;
			return sorted[idx];
		}
​
		// Decimación determinista por stride con duplicación al llegar al cap
		private void AddSample(List<double> list, double v, ref int stride, ref long seen)
		{
			seen++;
			if (seen % stride != 0) return;
			if (list.Count >= CalibSampleCap)
			{
				List<double> half = new List<double>(list.Count / 2 + 1);
				for (int i = 1; i < list.Count; i += 2) half.Add(list[i]);
				list.Clear(); list.AddRange(half);
				stride *= 2;
			}
			list.Add(v);
		}

		#region Motor de rachas
		private void ResetStreak()
		{
			_dir = 0; _streak = 0; _validSteps = 0; _fails = 0;
			_swH = 0; _swL = 0; _totalVol = 0; _maxRetroTicks = 0;
			_msList.Clear();
		}
​
		private void StartStreak(int dir, double price, double vol, DateTime t)
		{
			_dir = dir; _streak = 1; _validSteps = 1; _fails = 0;
			_swH = price; _swL = price; _totalVol = vol; _maxRetroTicks = 0;
			_msList.Clear();
			_tStart = t; _tLast = t;
		}
​
		private void ContinueStreak(double price, double vol, double ms, DateTime t, bool valid)
		{
			_streak++;
			if (valid) _validSteps++;
			if (price > _swH) _swH = price;
			if (price < _swL) _swL = price;
			_msList.Add(ms);   // FIX#2: solo intervalos internos
			_totalVol += vol;
			_tLast = t;
		}
​
		private void StepEngine(double price, double vol, double ms, DateTime t)
		{
			double prev = Closes[1][1];
​
			// (b) pausa: corta la racha; NO re-siembra a través de la pausa
			if (_dir != 0 && ms > _effMaxPausaMs)
			{
				FinalizeStreak(t);
				return;
			}
​
			if (_dir == 0)
			{
				// No iniciar en ticks planos (corrige sesgo isDown-first de v1)
				if      (price < prev) StartStreak(-1, price, vol, t);
				else if (price > prev) StartStreak(+1, price, vol, t);
				return;
			}
​
			bool valid = (_dir == 1) ? price >= prev : price <= prev;
			if (valid)
			{
				ContinueStreak(price, vol, ms, t, true);
				_fails = 0;
			}
			else
			{
				_fails++;
				if (_fails <= FallosTolerados)
					ContinueStreak(price, vol, ms, t, false);
				else
				{
					// (a) fallos agotados: finalizar y re-sembrar con el tick actual
					FinalizeStreak(t);
					if      (price < prev) StartStreak(-1, price, vol, t);
					else if (price > prev) StartStreak(+1, price, vol, t);
					return;
				}
			}
​
			// (c) retroceso relativo (V3): allowed = max(piso, pct% de la altura)
			if (UseRelativeRetro && _dir != 0)
			{
				double heightTicks = (_swH - _swL) / TickSize;
				double retro = _dir == 1 ? (_swH - price) / TickSize : (price - _swL) / TickSize;
				if (retro > _maxRetroTicks) _maxRetroTicks = retro;
				double allowed = Math.Max(RetroFloorTicks, RetroPctHeight / 100.0 * heightTicks);
				if (retro > allowed)
					FinalizeStreak(t);   // el tick que viola queda incluido (declarado)
			}
		}
​
		private void FinalizeStreak(DateTime t)
		{
			if (_dir == 0) { ResetStreak(); return; }
​
			int    heightTicks = (int)Math.Round((_swH - _swL) / TickSize);
			bool   isSweep     = heightTicks >= MinSweepTicks;   // FIX#3
			double totalMs     = 0;
			for (int i = 0; i < _msList.Count; i++) totalMs += _msList[i];
			double avgMs   = _msList.Count > 0 ? totalMs / _msList.Count : 0;
			double durSec  = Math.Max(totalMs, 1.0) / 1000.0;    // FIX#1
			double volRate = _totalVol / durSec;
​
			int  minReq     = isSweep ? MinPasos : MinAbsorbPasos;
			bool structural = _validSteps >= minReq && (isSweep || DetectAbsorb);
			bool pass = structural
				&& avgMs    <= _effMaxAvgMs
				&& totalMs  <= _effMaxTotalMs
				&& volRate  >= _effMinVolRate
				&& _totalVol >= _effMinTotalVol;
​
			// OBS: piso de export para fuerza bruta offline (con flag pass)
			if (_validSteps >= MinExportValidSteps && _log != null)
			{
				Zone obs = BuildZone(heightTicks, isSweep, avgMs, totalMs, volRate, assignId: false);
				LogRow("OBS", _tLast, obs, pass ? "pass=1" : "pass=0", "");
			}
​
			if (pass)
			{
				Zone z = BuildZone(heightTicks, isSweep, avgMs, totalMs, volRate, assignId: true);
				_zones.Add(z);
				LogRow("ZONE_CREATED", _tLast, z, "", "");
			}
​
			ResetStreak();
		}
​
		private Zone BuildZone(int heightTicks, bool isSweep, double avgMs, double totalMs, double volRate, bool assignId)
		{
			HZ2Bucket bucket = !isSweep            ? HZ2Bucket.Absorb
				: avgMs <= _effPredMs              ? HZ2Bucket.Predator
				: avgMs <= _effUltraMs             ? HZ2Bucket.Ultra
				:                                     HZ2Bucket.Fast;
​
			Zone z = new Zone();
			z.Id                = assignId ? ++_nextZoneId : 0;
			z.CalibId           = _calibId;
			z.CreatedPrimaryBar = (CurrentBars != null && CurrentBars.Length > 0) ? CurrentBars[0] : 0;
			z.CreatedTime       = _tLast;
			z.Direction         = _dir;
			z.Bucket            = bucket;
			z.AvgMs             = avgMs;
			z.TotalMs           = totalMs;
			z.VolRate           = volRate;
			z.TotalVol          = _totalVol;
			z.MaxRetroTicks     = _maxRetroTicks;
			z.Pasos             = _streak;
			z.ValidSteps        = _validSteps;
			z.HeightTicks       = heightTicks;
			// Geometría declarada: soporte bajo el origen alcista / resistencia sobre el bajista
			if (_dir == 1) { z.Upper = _swL;                                     z.Lower = _swL - ZoneHeightTicks * TickSize; }
			else           { z.Upper = _swH + ZoneHeightTicks * TickSize;        z.Lower = _swH; }
			z.Tag = "HZ2_" + z.Id;
			z.NeedsRedraw = true;
			return z;
		}

		#region Ciclo de vida de zonas (tick-exacto)
		private void UpdateZonesWithTick(double price, DateTime t)
		{
			for (int i = _zones.Count - 1; i >= 0; i--)
			{
				Zone z = _zones[i];
				if (z.Archived) continue;
​
				bool inside = price >= z.Lower && price <= z.Upper;
				if (inside && !z.InsideEpoch)
				{
					z.Touches++;
					z.InsideEpoch = true;
					z.NeedsRedraw = true;
					if (z.Touches <= MaxLoggedTouches)
						LogRow("ZONE_TOUCHED", t, z, "epoch=" + z.Touches, "");
​
					if (InvalidationMode == HZ2InvalidationMode.FirstTouch)
					{ InvalidateZone(z, t, "first_touch"); continue; }
					if (MaxTouches > 0 && z.Touches >= MaxTouches)
					{ InvalidateZone(z, t, "max_touches"); continue; }
				}
				else if (!inside)
					z.InsideEpoch = false;
​
				if (InvalidationMode == HZ2InvalidationMode.CloseThrough)
				{
					bool through = z.Direction == 1
						? price <= z.Lower - PenetrationTicks * TickSize
						: price >= z.Upper + PenetrationTicks * TickSize;
					if (through) InvalidateZone(z, t, "close_through");
				}
			}
		}
​
		private void InvalidateZone(Zone z, DateTime t, string reason)
		{
			z.Archived = true;
			z.NeedsRedraw = true;
			LogRow("ZONE_INVALIDATED", t, z, reason, "");
		}
​
		private void ExpireZones()
		{
			for (int i = _zones.Count - 1; i >= 0; i--)
			{
				Zone z = _zones[i];
				if (z.Archived) continue;
				if (CurrentBar - z.CreatedPrimaryBar <= MaxAgeBars) continue;
				z.Archived = true;
				z.NeedsRedraw = true;
				LogRow("ZONE_EXPIRED", Time[0], z, "", "");
			}
		}

		#region Render (solo visual, solo hilo primario)
		private void DrawZones()
		{
			if (!ShowZones) { PruneArchived(); return; }
​
			int drawn = 0;
			for (int i = _zones.Count - 1; i >= 0; i--)
			{
				Zone z = _zones[i];
				if (z.Id == 0) continue;
				if (z.Archived && !z.NeedsRedraw) continue;
				if (!z.Archived && drawn >= MaxZonesDrawn) continue;
​
				int startBA = CurrentBar - z.CreatedPrimaryBar;
				if (startBA < 0 || startBA > CurrentBar) { z.NeedsRedraw = false; continue; }
​
				Brush fill = z.Bucket == HZ2Bucket.Absorb ? ColorAbsorb
					: z.Direction == 1 ? ColorSupport : ColorResistance;
				Brush border = z.Archived ? Brushes.DimGray
					: z.Bucket == HZ2Bucket.Predator ? Brushes.Magenta
					: z.Bucket == HZ2Bucket.Ultra    ? Brushes.Orange
					: Brushes.Transparent;
​
				Draw.Rectangle(this, z.Tag, false, startBA, z.Upper, 0, z.Lower, border, fill, RectOpacity);
				z.NeedsRedraw = false;
				if (!z.Archived) drawn++;
			}
			PruneArchived();
		}
​
		private void PruneArchived()
		{
			for (int i = _zones.Count - 1; i >= 0; i--)
				if (_zones[i].Archived && !_zones[i].NeedsRedraw)
					_zones.RemoveAt(i);
		}

		private void CloseLog()
		{
			try
			{
				if (_log == null) return;
				DateTime t = (_tLast != DateTime.MinValue) ? _tLast : DateTime.Now;
				foreach (Zone z in _zones)
					if (!z.Archived && z.Id > 0) LogRow("SESSION_END", t, z, "snapshot", "");
				_log.Flush(); _log.Close(); _log.Dispose(); _log = null;
				Print("[HFTZones2] Log cerrado.");
			}
			catch (Exception ex) { Print("[HFTZones2] Error cerrando log: " + ex.Message); }
		}

		private void LogRow(string type, DateTime t, Zone z, string reason, string extra)
		{
			if (_log == null) return;
			try
			{
				CultureInfo inv = CultureInfo.InvariantCulture;
				StringBuilder sb = new StringBuilder(256);
				sb.Append(++_seq).Append(',');
				sb.Append(type).Append(',');
				sb.Append(t.ToString("yyyy-MM-dd HH:mm:ss.fff", inv)).Append(',');
				sb.Append(new DateTimeOffset(t.ToUniversalTime()).ToUnixTimeMilliseconds()).Append(',');
				if (z != null)
				{
					sb.Append(z.Id > 0 ? "Z" + z.Id.ToString("D6", inv) : "").Append(',');
					sb.Append(z.CalibId.ToString(inv)).Append(',');
					sb.Append(z.Direction.ToString(inv)).Append(',');
					sb.Append(z.Bucket.ToString().ToUpperInvariant()).Append(',');
					sb.Append(z.Upper.ToString("F6", inv)).Append(',');
					sb.Append(z.Lower.ToString("F6", inv)).Append(',');
					sb.Append(z.HeightTicks.ToString(inv)).Append(',');
					sb.Append(z.Pasos.ToString(inv)).Append(',');
					sb.Append(z.ValidSteps.ToString(inv)).Append(',');
					sb.Append(z.AvgMs.ToString("F3", inv)).Append(',');
					sb.Append(z.TotalMs.ToString("F1", inv)).Append(',');
					sb.Append(z.VolRate.ToString("F2", inv)).Append(',');
					sb.Append(z.TotalVol.ToString("F2", inv)).Append(',');
					sb.Append(z.MaxRetroTicks.ToString("F1", inv)).Append(',');
					sb.Append(z.Touches.ToString(inv)).Append(',');
				}
				else
					sb.Append(",,,,,,,,,,,,,,,");
				sb.Append(reason ?? "").Append(',');
				sb.Append(extra ?? "");
				_log.WriteLine(sb.ToString());
​
				if (type == "CALIBRATION" || type == "ZONE_INVALIDATED" || type == "ZONE_EXPIRED" || type == "SESSION_END" || ++_sinceFlush >= 500)
				{ _log.Flush(); _sinceFlush = 0; }
			}
			catch (Exception ex) { Print("[HFTZones2] Error escribiendo log: " + ex.Message); }
		}
