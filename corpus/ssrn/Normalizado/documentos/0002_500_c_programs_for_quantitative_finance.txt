Abstract
I have put together a compendium of over 500 C++ programs, discussing a plethora of topics spanning from modern C++ programming style, the standard template library(STL) containers and algorithms, numerical linear algebra, using Quantlib, to creating data structures and solvers commonly
found in quantitative ﬁnance. The programs are intended to be descriptive and pedagogical.

//The first version of a discount_curve might consist of
//a set of zero rates corresponding to key tenors, and additionally
//keeping track of today & the total number points.
struct discount_curve
{
date *t;
double* rate;
int n_points;
date today;
};
//A variable of the type discount curve can be defined as
//discount_curve disc;
//However, by itself that is not of much use, because disc's
//t and rate pointers do not point to anything. To be useful, we
//must give disc some elements to point to. For example, we can construct
//a discount_curve like this:
void init_discount_curve(discount_curve& d, int n, date today)
{
d.n_points = n;
d.today = today;
d.t = new date[n];
d.rate = new double[n];
}
//That is, disc's t member gets a pointer to the memory allocated for
//the block of dates by the new operator. Likewise for rate.
//n_points member is initialized to the number of points n. today
//member is initialized to today's date.
//The new operator allocates member from an area called the free store.
//A simple routine to input the key zero rates for various dates

double disc_factor(discount_curve& d, date from_date, date to_date)
{
date a, b, x, y;
double r_a, r_b, r_x, r_y;
if (to_date == from_date)
return 1.0;
//Search the tenor bucket [a,b) of the from_date
for (int i = 1; i < d.n_points; ++i)
{
if (from_date >= d.t[i - 1] && from_date < d.t[i])
{
a = d.t[i - 1];
b = d.t[i];
r_a = d.rate[i - 1];
r_b = d.rate[i];
break;
}
}
//Search the tenor bucket [x,y) of the to_date
for (int i = 1; i < d.n_points; ++i)
{
if (to_date >= d.t[i - 1] && to_date < d.t[i])
{
x = d.t[i - 1];
y = d.t[i];
r_x = d.rate[i - 1];
r_y = d.rate[i];
break;
}
}
double r1 = linear_interpolation(d.today, b, from_date, r_a, r_b);
double r2 = linear_interpolation(d.today, y, to_date, r_x, r_y);
double T1 = (from_date - d.today).days()/365;
double df1 = 1 / pow((1 + r1), T1);
double T2 = (to_date - d.today).days()/365;
